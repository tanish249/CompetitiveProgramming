b=int(input())  
 
count = 0 
total = 0 
 
while count<b:
    a=input()
    if a=="Tetrahedron":
        total +=4
    elif a=="Cube":
        total +=6
    elif a=="Octahedron":
        total +=8
    elif a=="Dodecahedron":
        total +=12
    elif a=="Icosahedron":
        total +=20
     
    count +=1
print(total)
